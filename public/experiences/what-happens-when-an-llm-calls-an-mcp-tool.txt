        { title: 'Guard', boundary: 'Middleware → validator', copy: 'Authentication, authorization, schema validation, and policy checks run before the side effect.', message: 'Bearer credential ✓

human-review policy ✓' },
        { title: 'Execute', boundary: 'Tool → application code', copy: 'Ordinary Python creates a branch, commits two validated files, and asks GitHub to open a pull request.', message: 'create branch

open pull request' },
        { title: 'Prove', boundary: 'Server → client → user', copy: 'A structured result gives the caller durable evidence and preserves a review boundary.', message: '{
  "number": 17,
  "url": "…/pull/17",
  "status": "open"
}' }
      ];
      const failures = [
        { label: 'Malformed JSON-RPC', tag: 'Protocol error', title: 'Reject before tool dispatch', copy: 'The request cannot be correlated or interpreted as a valid JSON-RPC operation.', code: '{ "jsonrpc": "2.0", "error": { "code": -32600 } }' },
        { label: 'Invalid post slug', tag: 'Validation error', title: 'Return repairable feedback', copy: 'The tool exists, but its arguments do not satisfy the published input contract.', code: 'slug: String should match pattern ^[a-z0-9]+…$' },
        { label: 'GitHub denies write', tag: 'Execution error', title: 'Report application failure', copy: 'Dispatch and validation succeeded; the downstream operation did not.', code: 'tool result: isError = true · permission denied' },
        { label: 'Pull request opened', tag: 'Structured result', title: 'Return evidence', copy: 'The client receives stable fields it can display or pass to a status tool.', code: '{ "number": 17, "status": "open", "url": "…" }' }
      ];
      const stageRoot = document.getElementById('stages');
      const boundary = document.getElementById('boundary');
      const title = document.getElementById('step-title');
      const copy = document.getElementById('step-copy');
      const message = document.getElementById('message');
      const position = document.getElementById('position');
      let active = 0;

      steps.forEach((step, index) => {
        const button = document.createElement('button');
        button.type = 'button';
        button.className = 'stage';
        button.innerHTML = '<span class="stage-number">0' + (index + 1) + '</span>' + step.title;
        button.addEventListener('click', () => renderStep(index, true));
        stageRoot.append(button);
      });

      function renderStep(index, focus) {
        active = Math.max(0, Math.min(steps.length - 1, index));
        const step = steps[active];
        const buttons = [...stageRoot.querySelectorAll('button')];
        buttons.forEach((button, buttonIndex) => {
          if (buttonIndex === active) button.setAttribute('aria-current', 'step');
          else button.removeAttribute('aria-current');
        });
        message.classList.add('refreshing');
        boundary.textContent = step.boundary;
        title.textContent = step.title;
        copy.textContent = step.copy;
        message.textContent = step.message;
        position.textContent = (active + 1) + ' / ' + steps.length;
        requestAnimationFrame(() => message.classList.remove('refreshing'));
        if (focus) buttons[active].focus();
      }

      document.getElementById('previous').addEventListener('click', () => renderStep(active - 1, true));
      document.getElementById('next').addEventListener('click', () => renderStep(active + 1, true));
      document.getElementById('reset').addEventListener('click', () => renderStep(0, true));
      document.addEventListener('keydown', event => {
        if (event.key === 'ArrowRight') renderStep(active + 1, true);
        if (event.key === 'ArrowLeft') renderStep(active - 1, true);
        if (event.key === 'Home') renderStep(0, true);
      });

      const casesRoot = document.getElementById('cases');
      const outcomeTag = document.getElementById('outcome-tag');
      const outcomeTitle = document.getElementById('outcome-title');
      const outcomeCopy = document.getElementById('outcome-copy');
      const outcomeCode = document.getElementById('outcome-code');
      failures.forEach((failure, index) => {
        const button = document.createElement('button');
        button.type = 'button';
        button.className = 'case';
        button.textContent = failure.label;
        button.addEventListener('click', () => renderFailure(index));
        casesRoot.append(button);
      });
      function renderFailure(index) {
        const failure = failures[index];
        [...casesRoot.querySelectorAll('button')].forEach((button, buttonIndex) => button.setAttribute('aria-pressed', String(buttonIndex === index)));
        outcomeTag.textContent = failure.tag;
        outcomeTitle.textContent = failure.title;
        outcomeCopy.textContent = failure.copy;
        outcomeCode.textContent = failure.code;
      }

      const themeControl = document.getElementById('theme-control');
      themeControl.addEventListener('click', () => {
        const root = document.documentElement;
        root.dataset.theme = root.dataset.theme === 'dark' ? 'light' : 'dark';
        themeControl.textContent = 'Theme: ' + root.dataset.theme;
      });
      new MutationObserver(() => {
        themeControl.textContent = 'Theme: ' + document.documentElement.dataset.theme;
      }).observe(document.documentElement, { attributes: true, attributeFilter: ['data-theme'] });

      renderStep(0, false);
      renderFailure(0);
    })();
  </script>
